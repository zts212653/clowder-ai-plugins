rows=[]
def run(stage,cmd,cwd):
    start=time.monotonic()
    with (OUT/f'{stage}.log').open('wb') as log:
        result=subprocess.run(cmd,cwd=cwd,env=ENV,stdout=log,stderr=subprocess.STDOUT)
    row={'stage':stage,'exitCode':result.returncode,'elapsedMs':round((time.monotonic()-start)*1000),'command':cmd,'cwd':str(cwd)}
    rows.append(row);(OUT/'stage-results.json').write_text(json.dumps(rows,indent=2)+'\n');print(json.dumps(row),flush=True)
    if result.returncode:sys.exit(result.returncode)
    return OUT/f'{stage}.log'

def update(v):
    if isinstance(v,list):return any(update(child) for child in v)
    if not isinstance(v,dict):return False
    if v.get('pluginId')=='official.github-operations' and 'versions' in v:
        row=next(row for row in v['versions'] if row['version']==metadata['version'])
        row['artifact'].update({key:metadata[key] for key in ['integrity','shasum']});return True
    return any(update(child) for child in v.values())
assert update(catalog)
